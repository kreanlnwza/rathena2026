prontera,164,174,4	script	Drop Rate Checker	857,{
	mes "[Drop Rate Checker]";
	mes "สวัสดีครับ! ผมสามารถตรวจสอบเรทดรอป";
	mes "ของแต่ละแผนที่ให้คุณได้";
	mes " ";
	mes "^FF0000เรทดรอปจะเปลี่ยนทุกๆ 3 ชั่วโมง^000000";
	mes "(00:00, 03:00, 06:00, 09:00, ...)";
	next;

	// แสดงเรทดรอปทั้งหมด
	mes "[รายการเรทดรอปแต่ละแผนที่]";
	mes "^00AAFF======================^000000";

	if ($map_count == 0) {
		mes "ยังไม่มีการกำหนดเรทดรอปสำหรับแผนที่ใดๆ";
		close;
	}

	// แสดงเรทดรอปแต่ละแผนที่
	.@found = 0;
	for (.@i = 0; .@i < $map_count; .@i++) {
		// ดึงเรทดรอปจากตัวแปร global
		.@rate = $map_rate[.@i];

		// ถ้าไม่ได้กำหนดไว้ ให้แสดง 100%
		if (.@rate == 0)
			.@rate = 100;

		// แสดงเฉพาะแผนที่ที่มีการปรับเปลี่ยนเรท (ไม่เท่ากับ 100%)
		if (.@rate != 100) {
			mes $map_name$[.@i] + ": ^FF0000" + .@rate + "%^000000";
			.@found++;
		}
	}

	if (.@found == 0)
		mes "ไม่มีแผนที่ที่มีเรทพิเศษในขณะนี้";

	mes "^00AAFF======================^000000";
	mes " ";
	mes "เรทดรอปจะถูกสุ่มใหม่ทุกๆ 3 ชั่วโมง";
	mes "ช่วงเรท: 150% - 300%";
	mes " ";
	mes "GM สามารถใช้คำสั่ง @droprate เพื่อสุ่มทันที";
	close;

OnInit:
	end;
}

// คำสั่ง @droprate สำหรับสุ่มเรทดรอปทันที
// คำสั่ง @droprateoff สำหรับปิด Event
-	script	DropRateCommand	-1,{
OnInit:
	bindatcmd "droprate", strnpcinfo(0)+"::OnCommand", 99, 99;
	bindatcmd "droprateoff", strnpcinfo(0)+"::OnCommandOff", 99, 99;
	end;

OnCommand:
	if (getgmlevel() < 99) {
		message strcharinfo(0), "GM Only";
		end;
	}
	donpcevent "DropRateAutoRandomize::OnRandomize";
	message strcharinfo(0), "Drop Rate Randomized!";
	end;

OnCommandOff:
	if (getgmlevel() < 99) {
		message strcharinfo(0), "GM Only";
		end;
	}
	donpcevent "DropRateAutoRandomize::OnDisable";
	message strcharinfo(0), "Drop Rate Event Disabled!";
	end;
}

	// ใช้รายชื่อแผนที่จาก OnInit ($map_name$)

	// รีเซ็ตเรททุกแผนที่เป็น 100% ก่อน
	for (.@i = 0; .@i < .@map_count; .@i++) {
		$map_rate[.@i] = 100;
		removemapflag($map_name$[.@i], mf_droprate);
	}

	// สุ่มจำนวนแผนที่ที่จะได้รับเรทพิเศษ (3-5 แผนที่)
	.@num_bonus = rand(3, 5);

	debugmes "[Drop Rate Auto Randomizer] Selecting " + .@num_bonus + " maps for bonus rates";

	// สร้างตัวแปรเพื่อเก็บ index ที่ถูกเลือกแล้ว
	setarray .@selected[0], -1, -1, -1, -1, -1;

	// สุ่มเลือกแผนที่
	for (.@i = 0; .@i < .@num_bonus; .@i++) {
		.@selected_index = 0;
		.@is_duplicate = 0;

		// วนลูปเพื่อสุ่มจนกว่าจะได้ index ที่ไม่ซ้ำ
		do {
			.@is_duplicate = 0;
			.@selected_index = rand(0, .@map_count - 1);

			// ตรวจสอบว่า index นี้ถูกเลือกไปแล้วหรือยัง
			for (.@j = 0; .@j < .@i; .@j++) {
				if (.@selected[.@j] == .@selected_index) {
					.@is_duplicate = 1;
					break;
				}
			}
		} while (.@is_duplicate);

		.@selected[.@i] = .@selected_index;

		// สุ่มเรทดรอปสำหรับแผนที่ที่ถูกเลือก
		.@map_rate = rand(150, 300);
		$map_rate[.@selected_index] = .@map_rate;

		// กำหนด mapflag
		setmapflag($map_name$[.@selected_index], mf_droprate, .@map_rate);
		debugmes "[Drop Rate Auto Randomizer] Set " + $map_name$[.@selected_index] + " to " + .@map_rate + "%";

		// เก็บชื่อแผนที่ที่ถูกเลือกไว้แสดงผล
		.@bonus_maps$[.@i] = $map_name$[.@selected_index] + " (" + .@map_rate + "%)";
	}

	// สร้างข้อความประกาศ
	.@announce$ = "แผนที่ที่ได้รับเรทพิเศษ: ";
	for (.@i = 0; .@i < .@num_bonus; .@i++) {
		if (.@i > 0)
			.@announce$ = .@announce$ + ", ";
		.@announce$ = .@announce$ + .@bonus_maps$[.@i];
	}

	// ประกาศในเซิร์ฟเวอร์
	announce "[Event] Drop Rate Randomizer: " + .@announce$, bc_all;
	end;

OnDisable:
	for (.@i = 0; .@i < $map_count; .@i++) {
		removemapflag($map_name$[.@i], mf_droprate);
		$map_rate[.@i] = 100;
