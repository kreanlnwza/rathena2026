//===== rAthena Script =======================================
//= Storage & Save Point Manager NPC
//===== Description: =========================================
//= Provides multiple storage options:
//= 1. Save Point - Save respawn location at prontera,155,179 (Free)
//= 2. Normal Storage - Access standard storage (800 Zeny)
//= 3. Type-based Storage - Item-type specific storages:
//=    Costume, Etc, Armor, Weapon, Card, Shadow Gear (25,000 Zeny each)
//= 4. Guild Storage - Guild warehouse access (25,000 Zeny)
//=    Requires guild membership and GD_GUILD_STORAGE skill
//=
//= NOTE: Requires premium storage IDs 9, 12-15, 19 configured in
//=       inter_server.yml and corresponding SQL tables.
//===== Changelogs: ==========================================
//= 1.0 First version - Costume Storage only
//= 2.0 Added item-type based storages (ID 10-19)
//= 3.0 Added Normal Storage, Save Point, and Guild Storage menu
//= 3.0 Maintained by V!be Coding [kreanlnwza] AI Assistant (Claude Sonnet 4.6)
//============================================================

prontera,150,180,4	script	Kafra Employee	4_F_KAFRA1,{
	.@cost = 25000;

	cutin "kafra_05",2;
	mes "[Kafra Employee]";
	mes "Welcome~!";
	mes "The Kafra Corporation";
	mes "is always working to provide";
	mes "you with convenient services.";
	mes "How may I be of assistance?";
	next;

	// Main menu
	switch(select("Save Point:Storage:Cancel")) {
		case 1: goto L_Save;
		case 2: goto L_StorageMenu;
		case 3:
			mes "[Kafra Employee]";
			mes "Come back anytime!";
			close2;
			cutin "kafra_05",-1;
			end;
	}

L_StorageMenu:
	mes "[Kafra Employee]";
	mes "Which storage would you like to access?";
	next;
	switch(select("Normal Storage:Type-based Storage:Guild Storage:Cancel")) {
		case 1: goto L_NormalStorage;
		case 2: goto L_TypeStorage;
		case 3: goto L_GuildStorage;
		case 4:
			mes "[Kafra Employee]";
			mes "Come back anytime!";
			close2;
			cutin "kafra_05",-1;
			end;
	}

L_NormalStorage:
	.@normal_cost = 800;
	mes "[Kafra Employee]";
	mes "Opening ^FF6347Normal Storage^000000...";
	mes "Fee: ^FF0000" + callfunc("F_InsertComma", .@normal_cost) + " Zeny^000000";
	next;

	if (Zeny < .@normal_cost) {
		mes "[Kafra Employee]";
		mes "Sorry, you don't have enough Zeny.";
		mes "You need ^FF0000" + callfunc("F_InsertComma", .@normal_cost) + " Zeny^000000.";
		mes "Your current Zeny: ^666666" + callfunc("F_InsertComma", Zeny) + " Zeny^000000.";
		close2;
		cutin "kafra_05",-1;
		end;
	}

	if (select("Pay and open storage:Cancel") == 2) {
		mes "[Kafra Employee]";
		mes "Come back anytime!";
		close2;
		cutin "kafra_05",-1;
		end;
	}

	Zeny -= .@normal_cost;
	dispbottom "Paid " + callfunc("F_InsertComma", .@normal_cost) + " Zeny for Normal Storage service.";
	close2;
	openstorage;
	end;

L_TypeStorage:
	mes "[Kafra Employee]";
	mes "Select item type storage:";
	next;
	switch(select("Costume Items:Etc Items:Armor:Weapon:Card:Shadow Gear:Back")) {
		case 1:  .@stor_id = 9;  .@stor_name$ = "Costume";      break;
		case 2:  .@stor_id = 12; .@stor_name$ = "Etc";           break;
		case 3:  .@stor_id = 13; .@stor_name$ = "Armor";         break;
		case 4:  .@stor_id = 14; .@stor_name$ = "Weapon";        break;
		case 5:  .@stor_id = 15; .@stor_name$ = "Card";          break;
		case 6:  .@stor_id = 19; .@stor_name$ = "Shadow Gear";   break;
		case 7:
			mes "[Kafra Employee]";
			mes "Come back anytime!";
			close2;
			cutin "kafra_05",-1;
			end;
	}
	goto L_Storage;

L_Save:
	save "prontera", 155, 179;	// Save Point
	mes "[Kafra Employee]";
	mes "^00FF00Save point updated!^000000";
	close2;
	cutin "kafra_05",-1;
	end;

L_GuildStorage:
	.@guildid = getcharid(2);

	if (.@guildid == 0) {
		mes "[Kafra Employee]";
		mes "Guild storage is only available for guild members.";
		close2;
		cutin "kafra_05",-1;
		end;
	}

	if (getgdskilllv(.@guildid, "GD_GUILD_STORAGE") == 0 || !guild_has_permission(GUILD_PERM_STORAGE)) {
		mes "[Kafra Employee]";
		mes "Your guild doesn't have guild warehouse skill or you don't have permission.";
		close2;
		cutin "kafra_05",-1;
		end;
	}

	mes "[Kafra Employee]";
	mes "Opening ^0055FFGuild Storage^000000...";
	mes "Fee: ^FF0000" + callfunc("F_InsertComma", .@cost) + " Zeny^000000";
	next;

	if (Zeny < .@cost) {
		mes "[Kafra Employee]";
		mes "Sorry, you don't have enough Zeny.";
		mes "You need ^FF0000" + callfunc("F_InsertComma", .@cost) + " Zeny^000000.";
		mes "Your current Zeny: ^666666" + callfunc("F_InsertComma", Zeny) + " Zeny^000000.";
		close2;
		cutin "kafra_05",-1;
		end;
	}

	if (select("Pay and open guild storage:Cancel") == 2) {
		mes "[Kafra Employee]";
		mes "Come back anytime!";
		close2;
		cutin "kafra_05",-1;
		end;
	}

	mes "[Kafra Employee]";
	mes "Opening the guild warehouse...";
	close2;

	if (Zeny < .@cost) {
		end;
	}

	if (guildopenstorage() == GSTORAGE_OPEN) {
		Zeny -= .@cost;
		dispbottom "Paid " + callfunc("F_InsertComma", .@cost) + " Zeny for Guild Storage service.";
		end;
	} else {
		dispbottom "Guild storage is currently in use by someone else. Please try again.";
		end;
	}

L_Storage:

	mes "[Kafra Employee]";
	mes "You selected: ^0000FF" + .@stor_name$ + " Storage^000000";
	mes "Only ^FF0000" + .@stor_name$ + "^000000 items can be stored here.";
	mes " ";
	mes "Fee: ^FF0000" + callfunc("F_InsertComma", .@cost) + " Zeny^000000";
	next;

	// Check Zeny
	if (Zeny < .@cost) {
		mes "[Kafra Employee]";
		mes "Sorry, you don't have enough Zeny.";
		mes "You need ^FF0000" + callfunc("F_InsertComma", .@cost) + " Zeny^000000.";
		mes "Your current Zeny: ^666666" + callfunc("F_InsertComma", Zeny) + " Zeny^000000.";
		close2;
		cutin "kafra_05",-1;
		end;
	}

	// Confirm
	switch(select("Pay and open storage:Cancel")) {
		case 1:
			Zeny -= .@cost;
			dispbottom "Paid " + callfunc("F_InsertComma", .@cost) + " Zeny for " + .@stor_name$ + " Storage service.";
			close2;
			openstorage2 .@stor_id,STOR_MODE_ALL;
			end;
		case 2:
			mes "[Kafra Employee]";
			mes "Come back anytime!";
			close2;
			cutin "kafra_05",-1;
			end;
	}
}
